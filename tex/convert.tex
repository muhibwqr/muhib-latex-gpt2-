\def\cv@inbs#1{\ifnum#1<33 0\else\ifnum#1<127 1\else\ifnum#1<161 0\else\ifnum#1=173 0\else1\fi\fi\fi\fi}
\cv@ub=0 \cv@un=0
\loop \ifnum\cv@inbs\cv@ub=1 \expandafter\xdef\csname cv@u\the\cv@ub\endcsname{\the\cv@ub}\else
  \expandafter\xdef\csname cv@u\the\numexpr256+\cv@un\endcsname{\the\cv@ub}\advance\cv@un1 \fi
  \ifnum\cv@ub<255 \advance\cv@ub1 \repeat
\def\cv@stream#1{\edef\cv@fn{#1}\cv@fsz=\pdffilesize{\cv@fn}\relax \cv@off=0 \cv@streamr}
\def\cv@streamr{\ifnum\cv@off<\cv@fsz
  \edef\cv@buf{\pdffiledump offset\cv@off length4096{\cv@fn}}\advance\cv@off4096
  \expandafter\cv@eatb\cv@buf\relax\relax \expandafter\cv@streamr\fi}
\def\cv@eatb#1#2{\ifx\relax#1\else \cv@ch="#1#2\relax \cv@state\expandafter\cv@eatb\fi}
\def\cv@addcp#1{\edef\cv@key{\cv@key.\the#1}\edef\cv@bl{\cv@bl\csname cv@u\the#1\endcsname,}}
\def\cv@addk#1{\edef\cv@key{\cv@key.\the#1}}
% vocab.json: {"key": id, ...} with \" \\ \uXXXX escapes, ids in file order
\def\cv@vs{\ifnum\cv@ch=34 \def\cv@key{}\def\cv@bl{}\let\cv@state\cv@vk\fi}
\def\cv@vk{\ifnum\cv@ch=92 \let\cv@state\cv@ve\else\ifnum\cv@ch=34 \cv@id=0 \let\cv@state\cv@vv\else\cv@addcp\cv@ch\fi\fi}
\def\cv@ve{\ifnum\cv@ch=117 \cv@cnt=0 \cv@cp=0 \let\cv@state\cv@vu\else \cv@addcp\cv@ch\let\cv@state\cv@vk\fi}
\def\cv@vu{\cv@cp=\numexpr16*\cv@cp+\ifnum\cv@ch<58 \cv@ch-48\else\ifnum\cv@ch<71 \cv@ch-55\else\cv@ch-87\fi\fi\relax
  \advance\cv@cnt1 \ifnum\cv@cnt=4 \cv@addcp\cv@cp\let\cv@state\cv@vk\fi}
\def\cv@vv{\ifnum\cv@ch>47 \ifnum\cv@ch<58 \cv@id=\numexpr10*\cv@id+\cv@ch-48\relax\fi\fi
  \ifnum\cv@ch=44 \cv@vend\fi\ifnum\cv@ch=125 \cv@vend\fi}
\def\cv@vend{\expandafter\xdef\csname cv@v\cv@key\endcsname{\the\cv@id}%
  \ifnum\cv@id=\cv@ida\else\errmessage{vocab.json ids out of order at \the\cv@id}\fi \advance\cv@ida1
  \immediate\write1{\BS!t{\ifnum\cv@id=50256 \else\cv@bl\fi}}\let\cv@state\cv@vs}
% merges.txt: "#version" line, then "a b" per line, UTF-8
\def\cv@mz{\ifnum\cv@ch=10 \def\cv@key{}\let\cv@state\cv@ma\fi}
\def\cv@piece#1#2{% #1 state for this piece, #2 action at its terminator
  \ifnum\cv@ch>191 \cv@cp=\numexpr\cv@ch-192\relax \let\cv@state\cv@mc \let\cv@ret#1\else\cv@addk\cv@ch\fi}
\def\cv@ma{\ifnum\cv@ch=32 \cv@ida=\csname cv@v\cv@key\endcsname\relax \def\cv@key{}\let\cv@state\cv@mb\else\cv@piece\cv@ma{}\fi}
\def\cv@mb{\ifnum\cv@ch=10 \cv@mend\else\cv@piece\cv@mb{}\fi}
\def\cv@mc{\cv@cp=\numexpr64*\cv@cp+\cv@ch-128\relax \cv@addk\cv@cp \let\cv@state\cv@ret}
\def\cv@mend{\immediate\write1{\BS!m{\the\cv@ida}{\csname cv@v\cv@key\endcsname}}\def\cv@key{}\let\cv@state\cv@ma}
\def\cv@tok{\immediate\openout1=\cvout \cv@vf.tex\relax \cv@ida=0 \let\cv@state\cv@vs \cv@stream{\cvdir vocab.json}%
  \immediate\closeout1 \cv@msg{vocab}%
  \immediate\openout1=\cvout \cv@mf.tex\relax \let\cv@state\cv@mz \cv@stream{\cvdir merges.txt}%
  \ifx\cv@state\cv@mb \ifx\cv@key\@empty\else\cv@mend\fi\fi \immediate\closeout1 \cv@msg{merges}}
\ifx\cv@vf\undefined\def\cv@vf{vocab}\def\cv@mf{merges}\fi
